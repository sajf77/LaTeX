\begin{theorem}
	If $f(z)$ is analytic in a simply connected domain $D$, and $\gamma$ is a closed rectifiable curve in $D$, then:
	\[
	\oint_{\gamma} f(z) \, dz = 0
	\]
\end{theorem}

\begin{theorem}
	Let $f(z)$ be analytic inside and on a simple closed contour $C$, oriented counterclockwise. If $z_0$ is any point in the interior of $C$, then:
	\[
	f(z_0) = \frac{1}{2\pi i} \oint_{C} \frac{f(z)}{z - z_0} \, dz
	\]
\end{theorem}

\begin{theorem}
	Every bounded entire function $f: \mathbb{C} \to \mathbb{C}$ must be constant. That is, if $f(z)$ is analytic for all $z \in \mathbb{C}$ and there exists a constant $M$ such that $|f(z)| \le M$, then $f(z) = C$ for all $z$.
\end{theorem}

\begin{theorem}
	Let $D$ be a simply connected open subset of the complex plane, and $f$ be a function analytic on $D$ except for a finite set of isolated singularities $\{z_1, z_2, \dots, z_n\}$. If $\gamma$ is a simple closed positively oriented curve in $D$ that does not pass through any singularity, then:
	\[
	\oint_{\gamma} f(z) \, dz = 2\pi i \sum_{k=1}^{n} \operatorname{Res}(f, z_k)
	\]
\end{theorem}
